\begin{frame}{MSA: core idea}
  \textbf{Multiple Sequence Alignment} was developed in bioinformatics to compare DNA / protein sequences.\\[0.1cm]
  \textbf{Idea}: insert gap characters ``\texttt{-}'' so that similar amino acids appear in the \emph{same column}.\\[0.1cm]

\begin{frame}[shrink=5]{Pairwise alignment: the scoring scheme}
  \textbf{Goal}: given two sequences, find an alignment (via gap insertion) that maximises the total score.
  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{Scoring function} $s(a, b)$:
      \begin{itemize}
        \item $s(a, a) = +m$ \quad \emph{match reward}
        \item $s(a, b) = -x$ \quad \emph{mismatch penalty} ($a \neq b$)
        \item $s(\cdot, \mathtt{-}) = -g$ \quad \emph{gap penalty}
      \end{itemize}
      \vspace{0.2cm}
      \textbf{Total alignment score} = sum of column-wise scores.\\[0.2cm]
      \textbf{Task}: find the alignment with the \emph{highest} total score — i.e., maximise the number of matches and minimise gaps/mismatches.
    \end{column}
    \begin{column}{0.51\textwidth}
      \textbf{Example} with $m{=}2,\; x{=}1,\; g{=}1$:\\[0.15cm]

\begin{frame}[shrink=5]{Needleman-Wunsch algorithm}
  \textbf{Needleman-Wunsch} (1970) finds the \emph{globally optimal} pairwise alignment in $O(|s_1| \cdot |s_2|)$ time.\\[0.3cm]
  \textbf{Recurrence} ($m{=}2,\; x{=}1,\; g{=}1$):
  \[
    F[i,j] = \max \begin{cases}
      F[i{-}1,\,j{-}1] + s(s_1^i,\, s_2^j) & \text{(diagonal)}\\
      F[i{-}1,\,j] - g                        & \text{(up)}\\
      F[i,\,j{-}1] - g                        & \text{(left)}
    \end{cases}
  \]
  \textbf{Initialisation}: $F[0,j] = -jg$,\quad $F[i,0] = -ig$\\[0.3cm]
  Each cell $F[i,j]$ stores the \emph{best score} for aligning the first $i$ characters of $s_1$ with the first $j$ characters of $s_2$.
\end{frame}

\begin{frame}[shrink=5]{From pairwise to multiple: profiles}
  \textbf{Problem}: NW works for 2 sequences. How to align $N > 2$ sequences jointly?\\[0.2cm]
  \textbf{Key idea}: represent an existing alignment as a \textbf{profile} -- a frequency matrix.
  \vspace{0.3cm}

  A \textbf{profile} $P$ is an $|\mathcal{A}| \times M$ matrix:
  \[
    P_{a,j} = \text{fraction of event type } a \text{ in column } j
  \]
  \textbf{Profile–sequence score}: to align sequence $s$ against profile $P$, replace the scalar score $s(a, b)$ with a weighted average:
  \[
    s_{\text{profile}}(s^i, P_j) = \sum_{a \in \mathcal{A}} P_{a,j} \cdot s(s^i, a)
  \]
  Then run NW on the profile as if it were a sequence.\\[0.2cm]
  {\footnotesize This extends naturally to aligning two profiles (profile--profile alignment).}
\end{frame}

\begin{frame}[shrink=5]{MSA: key properties}
  \vspace{0.6cm}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \textbf{Complexity}:
      \begin{itemize}
        \item \textbf{Two sequences}: $O(|s_1| \cdot |s_2|)$ time
        \item \textbf{Multiple sequences}: NP-hard -- exponential search over all gap placements
      \end{itemize}
      \vspace{0.25cm}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Limitations}:
      \begin{itemize}
        \item \textbf{Greedy}: early merge errors propagate — no backtracking once a column is aligned
        \item \textbf{Guide tree sensitivity}: different orderings of merges can produce different final alignments
        \item \textbf{Sub-optimal}: the progressive result is not guaranteed to maximise the total alignment score
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=5]{Aggregate tree: hierarchical clustering of sequences}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Input}: $N$ unique event sequences $S = \{s_1,\ldots,s_N\}$\\[0.2cm]
      \textbf{Goal}: build a tree that organises sequences into nested clusters — so the user can explore data at any level of granularity.\\[0.2cm]
      \textbf{Algorithm} (agglomerative / bottom-up):
      \begin{enumerate}
        \item Each $s_i$ starts as its own leaf node
        \item Compute pairwise distance matrix $d$:
              \[d[i,j] = D_{\text{qgram}}(s_i,\, s_j)\]
        \item Repeat until one root remains:
        \begin{enumerate}[a.]
          \item Merge the two closest nodes $n_a,\,n_b$
          \item Update $d$ (average linkage)
        \end{enumerate}
      \end{enumerate}
      \vspace{0.15cm}
      \textbf{Distance}: cosine distance of \textbf{q-gram profiles} — vectors of sub-sequence counts of length $q$; with $q{=}1$ order is ignored.
    \end{column}

%%------------------------------------------------------------
\begin{frame}[shrink=5]{Score: information score per column}
  After alignment, measure how \textbf{homogeneous} each column $j$ is:
  \[
    \boxed{I_j = 1 - \frac{E_j}{\log_2(|\mathcal{A}|+1)}}
  \]
  where $E_j$ is the \textbf{entropy} of event types in column $j$:
  \[
    E_j = \sum_{a \,\in\, \mathcal{A}_j \cup \{-\}} \begin{cases}
      -P_a \log_2\!\bigl(P_a / G_j\bigr), & a = \text{``-''} \\[4pt]
      -P_a \log_2 P_a,                     & \text{otherwise}
    \end{cases}
  \]
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $P_a$: probability (frequency) of event type $a$ in column $j$
        \item $G_j$: count of gaps in column $j$ (used in gap entropy term)
        \item $|\mathcal{A}|$: number of event types (alphabet size)
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Interpretation}
        \begin{itemize}
          \item Column all same event $\Rightarrow$ $E_j = 0$, $I_j = 1$
          \item Column all gaps $\Rightarrow$ $I_j \approx 0$
          \item Column random mix $\Rightarrow$ $I_j \approx 0$
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%%------------------------------------------------------------
\begin{frame}[shrink=5]{Simplify: merging low-information columns}

      \textbf{Rule}: merge columns $j$ and $j{+}1$ if \emph{both} $I_j < I_\tau$ and $I_{j+1} < I_\tau$.\\[0.2cm]
      \textbf{How merging works}:
      \begin{itemize}
        \item Concatenate characters $\lambda_{i,j}$ and $\lambda_{i,j+1}$ for each row $i$
        \item The merged element becomes a \emph{sub-sequence}
        \item Repeat until no two adjacent columns are both below $I_\tau$
      \end{itemize}

        \item Length $M' \leq M$

\begin{frame}[shrink=5]{Average silhouette width}
  \textbf{Goal}: automatically suggest a good number of clusters $k$ to the user.\\[0.2cm]
  For each sequence $s$ in cluster $C_s$, define the \textbf{silhouette value}:
  \[
    z(s) = \frac{v(s) - u(s)}{\max\bigl(u(s),\, v(s)\bigr)}, \qquad z(s) \in [-1,\, 1]
  \]
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{itemize}
        \item $u(s)$: mean distance from $s$ to other sequences in the \emph{same} cluster $C_s$
        \item $v(s)$: mean distance from $s$ to sequences in the \emph{nearest other} cluster
        \item $z(s) \approx +1$: $s$ is well-placed (compact, separated)
        \item $z(s) \approx -1$: $s$ would fit better in another cluster
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Average silhouette width} for $k$ clusters:
      \[
        \bar{z}(k) = \frac{1}{N}\sum_{s \in S} z(s)
      \]
      \begin{block}{Strategy}
        Find the $k$ that \emph{maximises} $\bar{z}(k)$.\\
        Also report \emph{local maxima} as alternative good clusterings.\\
        Users can explore all recommended $k$ values interactively.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=5]{Key contributions}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{What Sequen-C introduced}:
      \begin{enumerate}
        \item \textbf{Multilevel overview} technique for event sequences — adjustable in two independent dimensions ($k$ and $I_\tau$)
        \item \textbf{Align-Score-Simplify}: a novel cluster data representation using MSA + entropy-based information score + iterative column merging
        \item \textbf{Alternative optimal clusterings} suggested via average silhouette width
        \item \textbf{Sequen-C system}: four coordinated views for multilevel and detail-on-demand exploration
      \end{enumerate}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Limitations}
        \begin{itemize}
          \item MSA step is $O(nl^2)$: slow for very long sequences ($>500$ input sequences, avg.\ length $>10$)
          \item Optimal $I_\tau$ value is dataset-dependent
          \item Large alphabets ($>$20 event types) cause visual clutter
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{block}{Future work}
        \begin{itemize}
          \item Event type categorisation to speed up MSA
          \item Alternative alignment: longest common sub-sequence
          \item Applications beyond clinical domain
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=5]{Temporal event sequences}
    A \textbf{temporal event sequence} is an ordered list of typed events for one entity:
    \[
      s = \langle e_1,\, e_2,\, \ldots,\, e_T \rangle, \qquad e_i \in \mathcal{A}
    \]
    where $\mathcal{A}$ is a finite \textbf{event type alphabet}.
    \vspace{0.3cm}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \textbf{Application domains}:
        \begin{itemize}
          \item \textbf{Healthcare}: patient journeys through care\\
            {\footnotesize\texttt{Admitted $\to$ ED triage $\to$ X-ray $\to$ Discharged}}
          \item \textbf{Emergency services}: call pathways\\
            {\footnotesize\texttt{Call $\to$ Ambulance $\to$ Scene $\to$ Hospital}}
          \item \textbf{Web analytics}: clickstream sessions\\
            {\footnotesize\texttt{Search $\to$ Product $\to$ Cart $\to$ Checkout}}
          \item \textbf{Business processes}: workflow logs
        \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
        \textbf{Key properties}:
        \begin{itemize}
          \item Events come from a \emph{discrete} alphabet
          \item Order matters, but exact timing varies
          \item Datasets contain \emph{thousands} of sequences
          \item High variability: few sequences are identical
          \item Goal: find common and deviating pathways
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}
  
  %%------------------------------------------------------------
  \begin{frame}[shrink=5]{The visual analytics challenge}
    \begin{columns}[T]
      \begin{column}{0.54\textwidth}
        \textbf{Goal}: provide an overview that is \emph{simplified but informative}.\\[0.25cm]
        \textbf{Problems with existing approaches}:
        \begin{itemize}
          \item Fixed level of detail — cannot adjust granularity
          \item Require repeated zooming to see important details
          \item Patterns visible at one scale are invisible at another
          \item No support for exploring \emph{alternative} clusterings
        \end{itemize}
        \vspace{0.25cm}
        \textbf{Two independent ``zoom'' axes needed}:
        \begin{enumerate}
          \item \textbf{Vertical}: number of sequence clusters $k$
          \item \textbf{Horizontal}: simplification depth within each cluster
        \end{enumerate}
      \end{column}
      \begin{column}{0.43\textwidth}
        \begin{block}{Information-seeking mantra}
          \emph{``Overview first, zoom and filter, then details on demand''} — Shneiderman 1996
        \end{block}
        \vspace{0.3cm}
        {\small
        Magallanes et al.,\\
        \textbf{Sequen-C: A Multilevel Overview of Temporal Event Sequences},\\
        IEEE TVCG, 2021\\[0.2cm]
        \texttt{arXiv:2108.03043}}
      \end{column}
    \end{columns}
  \end{frame}

  \sectionframe{Building the Aggregate Tree}
  
  %%------------------------------------------------------------
  \begin{frame}[shrink=5]{Aggregate tree: bottom-up hierarchical clustering}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Input}: $N$ unique sequences $S = \{s_1,\ldots,s_N\}$\\[0.2cm]
        \textbf{Algorithm} (agglomerative):
        \begin{enumerate}
          \item Initialize $T$: each $s_i$ is a leaf node
          \item Compute pairwise distance matrix $d$:\\
                {\small $d[i,j] = D_{\mathrm{qgram}}(s_i, s_j)$}
          \item \textbf{Repeat} until $|T| = 1$:
          \begin{enumerate}[a.]
            \item Find closest pair $(n_a, n_b) = \operatorname{argmin}_{n_i,n_j} d[n_i, n_j]$
            \item Create $n_{a,b} = \textbf{aggregate}(n_a, n_b)$ \emph{(Align-Score-Simplify)}
            \item Update $d$ via average agglomeration
          \end{enumerate}
          \item Return tree $T$
        \end{enumerate}
        \vspace{0.2cm}
        \textbf{Distance}: cosine distance of \emph{q-gram profiles}\\
        {\small (vectors of sub-sequence counts of length $q$; $q{=}1$ groups sequences regardless of event order)}
      \end{column}
      \begin{column}{0.45\textwidth}
        \vspace{0.5cm}
        \begin{center}
