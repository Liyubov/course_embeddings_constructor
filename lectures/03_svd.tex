\sectionframe{1. Singular Value Decomposition (SVD)}

\begin{frame}{Motivating problem: Recommender Systems}
  A streaming platform knows which movies each user has watched and rated.\\[0.3cm]
  \textbf{Goal:} predict missing ratings — recommend movies the user hasn't seen yet.
  \vspace{0.3cm}
  \begin{center}
  \begin{tabular}{lcccccc}
    \toprule
    & \textit{Inception} & \textit{Titanic} & \textit{Alien} & \textit{La La Land} & \textit{Interstellar} \\
    \midrule
    Alice & 5 & 3 & 4 & 2 & \textcolor{accentred}{?} \\
    Bob   & 4 & 1 & 5 & 1 & 5 \\
    Carol & 2 & 5 & 1 & \textcolor{accentred}{?} & 2 \\
    Dave  & \textcolor{accentred}{?} & 4 & 2 & 5 & 3 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.3cm}
  This is a \textbf{user–item matrix} $X \in \mathbb{R}^{N \times p}$, mostly \emph{sparse}: most entries are missing.
\end{frame}

\begin{frame}{Structure in the user--item matrix}
  Users don't rate items independently — their preferences reflect \textbf{latent factors}:
  genres, styles, directors, mood, \ldots
  \vspace{0.3cm}
  \begin{itemize}
    \item Alice and Bob both like sci-fi $\Rightarrow$ their rows are correlated
    \item Inception and Interstellar are both Nolan sci-fi $\Rightarrow$ their columns are correlated
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Key insight:} although $X$ is $N \times p$, the \emph{effective} degrees of freedom are much smaller — the matrix is approximately \emph{low-rank}.\\[0.3cm]
  \begin{block}{Idea}
    Find a low-rank approximation $X \approx \hat X$ and use $\hat X_{ij}$ to fill in the missing entries.
  \end{block}
\end{frame}

\begin{frame}{SVD: Definition}
  Any matrix $X \in \mathbb{R}^{N \times p}$ can be decomposed as
  $$X = U \Sigma V^\top,$$
  where
  \begin{itemize}
    \item $U \in \mathbb{R}^{N \times N}$ — orthogonal ($U^\top U = I$), columns $u_j$ are left singular vectors
    \item $\Sigma \in \mathbb{R}^{N \times p}$ — diagonal matrix of singular values $\sigma_1 \geq \sigma_2 \geq \cdots \geq 0$
    \item $V \in \mathbb{R}^{p \times p}$ — orthogonal ($V^\top V = I$), columns $v_j$ are right singular vectors
  \end{itemize}
  
\end{frame}

\begin{frame}{Truncated SVD}
  Keep only the top $q$ singular values — \textbf{best rank-$q$ approximation} (Eckart–Young theorem):
  $$X \approx X_q = U_q \Sigma_q V_q^\top,$$
  where $U_q \in \mathbb{R}^{N\times q}$, $\Sigma_q \in \mathbb{R}^{q\times q}$, $V_q \in \mathbb{R}^{p\times q}$.
  \vspace{0.2cm}
  \begin{block}{Eckart--Young theorem}
    $X_q$ minimises $\|X - \hat X\|_F$ over all rank-$q$ matrices $\hat X$.
  \end{block}
  \vspace{0.2cm}
  \begin{itemize}
    \item Columns of $V_q$ are the same as principal components of PCA
    \item $U_q \Sigma_q$ gives the low-dimensional embedding (scores)
    \item Singular values $\sigma_k$ measure variance explained by each component
  \end{itemize}
\end{frame}

\begin{frame}{SVD vs PCA}
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    & \textbf{PCA} & \textbf{SVD} \\
    \midrule
    Input      & centered $X$ & any $X$ \\
    Decompose  & $X^\top X$ (covariance) & $X$ directly \\
    Components & eigenvectors of $X^\top X$ & right singular vectors $V$ \\
    Scores     & $XV$ & $U\Sigma$ \\
    Numerics   & less stable & more stable \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.3cm}
  PCA on centered $X$ $\equiv$ truncated SVD on centered $X$.
\end{frame}

\sectionframe{2. NMF decomposition}
\begin{frame}{Idea}
Assume $X \geq 0$. We want to find a decompositoin
$$X_{N\times p} \approx W_{N\times q}H_{q\times p},$$
where $w_{ik} \geq 0$, $h_{kj} \geq 0$.
\begin{itemize}
  \item Assume that $x_{ij} \sim Pois(\lambda_{ij})$, $\lambda_{ij}=(WH)_{ij}=\sum\limits_{k}w_{ik}h_{kj}$
  \item Likelihood $l(W, H) = \prod\limits_{i,j} \frac{e^{-\lambda_{ij}} \lambda_{ij}^{x_{ij}}}{x_{ij}!}$
  \item $\log l(W, H) = \underbrace{\sum\limits_{i,j} x_{ij}\log{\lambda_{ij}} - \lambda_{ij}}_{L(W, H)} - \log(x_{ij}!) \xrightarrow{w_{ij}, h_{ij}} \max$
\end{itemize}
\end{frame}

\begin{frame}{Optimization}
$$L(W, H) = \sum\limits_{i,j} x_{ij}\log{\lambda_{ij} - \lambda_{ij}} \xrightarrow{w_{ij}, h_{ij}} \max$$
\begin{itemize}
  \item $\frac{\partial L}{\partial w_{ik}} = \sum\limits_{j} \left(\frac{x_{ij}}{\lambda_{ij}}\frac{\partial \lambda_{ij}}{\partial w_{ik}} - \frac{\partial \lambda_{ij}}{\partial w_{ik}}\right) = \underbrace{\sum\limits_{j} \frac{x_{ij}}{\lambda_{ij}}h_{kj}}_{A_{ik}} - \underbrace{\sum\limits_{j} h_{kj}}_{B_{ik}}$,
  \item $\frac{\partial L}{\partial h_{kj}} = \sum\limits_{i} \left(\frac{x_{ij}}{\lambda_{ij}}\frac{\partial \lambda_{ij}}{\partial h_{kj}} - \frac{\partial \lambda_{ij}}{\partial h_{kj}}\right) = \sum\limits_{i} \frac{x_{ij}}{\lambda_{ij}} w_{ik} - \sum\limits_{i}w_{ik}$
  \item Due to $w_{ik}\geq 0, h_{kj}\geq 0$ we cannot simply use gradient descent.
  \item Lee and Seung 2001: 
\end{itemize}
$$w_{ik} \leftarrow w_{ik}\frac{A_{ik}}{B_{ik}} = \frac{\sum_{j} h_{kj}x_{ij} / (WH)_{ij}}{\sum_{j} h_{kj}}$$
$$h_{kj} \leftarrow w_{ik}\frac{C_{kj}}{D_{kj}} = \frac{\sum_{i} w_{ik}x_{ij} / (WH)_{ij}}{\sum_{i} w_{ik}}$$
\end{frame}
